// Command server runs the calculator REST API.
//
// Environment variables:
//
//        PORT            port to listen on (default 8080)
//        ALLOWED_ORIGIN  browser origin allowed by CORS (default http://localhost:5173)
//        STATIC_DIR      directory of built frontend files to serve (default: unset, API only)
package main

import (
        "context"
        "errors"
        "fmt"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "path/filepath"
        "syscall"
        "time"

        "github.com/Julianmotta07/sezzle-calculator/backend/internal/api"
)

// shutdownTimeout is how long in-flight requests get to finish on shutdown.
const shutdownTimeout = 10 * time.Second

func main() {
        if err := run(); err != nil {
                slog.Error("server stopped with error", "err", err)
                os.Exit(1)
        }
}

// run starts the server and blocks until it fails or receives SIGINT/SIGTERM,
// then shuts down gracefully.
func run() error {
        port := envOr("PORT", "8080")
        origin := envOr("ALLOWED_ORIGIN", "http://localhost:5173")
        staticDir := os.Getenv("STATIC_DIR")
        if staticDir != "" {
                // Fail fast on a misconfigured path instead of serving 404s at runtime.
                if _, err := os.Stat(filepath.Join(staticDir, "index.html")); err != nil {
                        return fmt.Errorf("STATIC_DIR %q has no index.html: %w", staticDir, err)
                }
        }

        srv := &http.Server{
                Addr:              ":" + port,
                Handler:           api.NewHandler(api.Config{AllowedOrigin: origin, StaticDir: staticDir}),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }

        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        serveErr := make(chan error, 1)
        go func() {
                slog.Info("server listening", "addr", srv.Addr, "allowed_origin", origin, "static_dir", staticDir)
                serveErr <- srv.ListenAndServe()
        }()

        select {
        case err := <-serveErr:
                // ListenAndServe only returns early on failure (e.g. port in use).
                return err
        case <-ctx.Done():
        }

        slog.Info("shutting down", "timeout", shutdownTimeout)
        shutdownCtx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return err
        }
        if err := <-serveErr; !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        slog.Info("server stopped")
        return nil
}

// envOr returns the environment variable key, or fallback if it is unset or empty.
func envOr(key, fallback string) string {
        if v := os.Getenv(key); v != "" {
                return v
        }
        return fallback
}

// Error codes returned in the "code" field of every error response.
// They are part of the public API contract, so clients can switch on them.
const (
        CodeInvalidJSON       = "INVALID_JSON"
        CodeUnknownField      = "UNKNOWN_FIELD"
        CodePayloadTooLarge   = "PAYLOAD_TOO_LARGE"
        CodeMissingOperation  = "MISSING_OPERATION"
        CodeUnknownOperation  = "UNKNOWN_OPERATION"
        CodeMissingOperand    = "MISSING_OPERAND"
        CodeUnexpectedOperand = "UNEXPECTED_OPERAND"
        CodeDivisionByZero    = "DIVISION_BY_ZERO"
        CodeNegativeSqrt      = "NEGATIVE_SQRT"
        CodeNonFiniteResult   = "NON_FINITE_RESULT"
        CodeNotFound          = "NOT_FOUND"
        CodeMethodNotAllowed  = "METHOD_NOT_ALLOWED"
        CodeInternal          = "INTERNAL_ERROR"
)

// ErrorResponse is the JSON shape shared by all error responses:
//
//        {"error":{"code":"DIVISION_BY_ZERO","message":"..."}}
type ErrorResponse struct {
        Error ErrorBody `json:"error"`
}

// ErrorBody holds the machine-readable code and human-readable message.
type ErrorBody struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

// apiError is an error that already knows its HTTP status and error code.
type apiError struct {
        status  int
        code    string
        message string
}

func (e *apiError) Error() string { return e.message }

// badRequest builds a 400 apiError.
func badRequest(code, message string) *apiError {
        return &apiError{status: http.StatusBadRequest, code: code, message: message}
}

// fromCalcError maps a calculator sentinel error to a 422 apiError.
// Unknown errors become a 500 so they are never silently misreported.
func fromCalcError(err error) *apiError {
        switch {
        case errors.Is(err, calculator.ErrDivisionByZero):
                return &apiError{http.StatusUnprocessableEntity, CodeDivisionByZero, "cannot divide by zero"}
        case errors.Is(err, calculator.ErrNegativeSqrt):
                return &apiError{http.StatusUnprocessableEntity, CodeNegativeSqrt, "cannot take the square root of a negative number"}
        case errors.Is(err, calculator.ErrNonFinite):
                return &apiError{http.StatusUnprocessableEntity, CodeNonFiniteResult, "result is not a finite number (overflow or undefined)"}
        default:
                slog.Error("unexpected calculator error", "err", err)
                return &apiError{http.StatusInternalServerError, CodeInternal, "internal server error"}
        }
}

// writeJSON writes v as a JSON response with the given status code.
func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        if err := json.NewEncoder(w).Encode(v); err != nil {
                slog.Error("failed to encode response", "err", err)
        }
}

// writeError writes e using the shared error shape.
func writeError(w http.ResponseWriter, e *apiError) {
        writeJSON(w, e.status, ErrorResponse{Error: ErrorBody{Code: e.code, Message: e.message}})
}

package api

import (
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "net/http"
        "reflect"
        "strings"
)

// maxBodyBytes caps the size of a calculate request body. A valid request is
// well under 200 bytes, so 4 KiB is generous while still bounding memory use.
const maxBodyBytes = 4 << 10

// calculateRequest is the body of POST /api/v1/calculate. Operands are
// pointers so a missing field can be told apart from an explicit 0.
type calculateRequest struct {
        Operation string   `json:"operation"`
        A         *float64 `json:"a"`
        B         *float64 `json:"b"`
}

// calculateResponse is the success body of POST /api/v1/calculate.
type calculateResponse struct {
        Operation string  `json:"operation"`
        Result    float64 `json:"result"`
}

// operationsResponse is the body of GET /api/v1/operations.
type operationsResponse struct {
        Operations []operation `json:"operations"`
}

// handleHealth reports that the server is up.
func handleHealth(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

// handleOperations lists the supported operations and their operand counts.
func handleOperations(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, operationsResponse{Operations: operations})
}

// handleCalculate decodes, validates and runs one calculation.
func handleCalculate(w http.ResponseWriter, r *http.Request) {
        req, apiErr := decodeCalculateRequest(w, r)
        if apiErr != nil {
                writeError(w, apiErr)
                return
        }
        op, apiErr := resolveOperation(req)
        if apiErr != nil {
                writeError(w, apiErr)
                return
        }

        var b float64
        if req.B != nil {
                b = *req.B
        }
        result, err := op.apply(*req.A, b)
        if err != nil {
                writeError(w, fromCalcError(err))
                return
        }
        writeJSON(w, http.StatusOK, calculateResponse{Operation: op.Name, Result: result})
}

// decodeCalculateRequest reads exactly one JSON object from a size-limited
// body, rejecting unknown fields and trailing data.
func decodeCalculateRequest(w http.ResponseWriter, r *http.Request) (calculateRequest, *apiError) {
        r.Body = http.MaxBytesReader(w, r.Body, maxBodyBytes)
        dec := json.NewDecoder(r.Body)
        dec.DisallowUnknownFields()

        var req calculateRequest
        if err := dec.Decode(&req); err != nil {
                return req, decodeError(err)
        }
        if err := dec.Decode(&struct{}{}); !errors.Is(err, io.EOF) {
                if tooLarge := asTooLarge(err); tooLarge != nil {
                        return req, tooLarge
                }
                return req, badRequest(CodeInvalidJSON, "request body must contain a single JSON object")
        }
        return req, nil
}

// decodeError turns a json.Decoder error into a client-facing apiError.
func decodeError(err error) *apiError {
        var syntaxErr *json.SyntaxError
        var typeErr *json.UnmarshalTypeError

        if tooLarge := asTooLarge(err); tooLarge != nil {
                return tooLarge
        }
        switch {
        case errors.Is(err, io.EOF):
                return badRequest(CodeInvalidJSON, "request body is empty")
        case errors.Is(err, io.ErrUnexpectedEOF):
                return badRequest(CodeInvalidJSON, "request body contains malformed JSON")
        case errors.As(err, &syntaxErr):
                return badRequest(CodeInvalidJSON, fmt.Sprintf("request body contains malformed JSON at position %d", syntaxErr.Offset))
        case errors.As(err, &typeErr):
                if typeErr.Field == "" {
                        return badRequest(CodeInvalidJSON, "request body must be a JSON object")
                }
                return badRequest(CodeInvalidJSON, fmt.Sprintf("field %q must be %s", typeErr.Field, describeKind(typeErr.Type)))
        case strings.HasPrefix(err.Error(), "json: unknown field "):
                // encoding/json has no typed error for this case; the prefix is stable.
                field := strings.TrimPrefix(err.Error(), "json: unknown field ")
                return badRequest(CodeUnknownField, "request body contains unknown field "+field)
        default:
                return badRequest(CodeInvalidJSON, "request body is not valid JSON")
        }
}

// asTooLarge returns a 413 apiError if err came from exceeding maxBodyBytes.
func asTooLarge(err error) *apiError {
        var maxErr *http.MaxBytesError
        if !errors.As(err, &maxErr) {
                return nil
        }
        return &apiError{
                status:  http.StatusRequestEntityTooLarge,
                code:    CodePayloadTooLarge,
                message: fmt.Sprintf("request body must not exceed %d bytes", maxBodyBytes),
        }
}

// describeKind names the JSON type a Go field expects, for error messages.
func describeKind(t reflect.Type) string {
        for t.Kind() == reflect.Pointer {
                t = t.Elem()
        }
        switch t.Kind() {
        case reflect.Float32, reflect.Float64:
                return "a finite number"
        case reflect.String:
                return "a string"
        default:
                return "of type " + t.String()
        }
}

// resolveOperation checks the operation name and that exactly the operands
// it needs are present.
func resolveOperation(req calculateRequest) (operation, *apiError) {
        if req.Operation == "" {
                return operation{}, badRequest(CodeMissingOperation, `field "operation" is required`)
        }
        op, ok := operationsByName[req.Operation]
        if !ok {
                return operation{}, badRequest(CodeUnknownOperation,
                        fmt.Sprintf("unknown operation %q; see GET /api/v1/operations", req.Operation))
        }
        if req.A == nil {
                return operation{}, badRequest(CodeMissingOperand,
                        fmt.Sprintf(`operation %q requires field "a"`, op.Name))
        }
        if op.Operands == 2 && req.B == nil {
                return operation{}, badRequest(CodeMissingOperand,
                        fmt.Sprintf(`operation %q requires field "b"`, op.Name))
        }
        if op.Operands == 1 && req.B != nil {
                return operation{}, badRequest(CodeUnexpectedOperand,
                        fmt.Sprintf(`operation %q takes only field "a"; remove field "b"`, op.Name))
        }
        return op, nil
}

// Package api exposes the calculator over a JSON REST API built on net/http.
package api

import "net/http"

// Config holds the settings the HTTP layer needs.
type Config struct {
        // AllowedOrigin is the single browser origin allowed by CORS,
        // for example "http://localhost:5173". Use "*" to allow any origin.
        AllowedOrigin string

        // StaticDir, when non-empty, is a directory of built frontend files
        // (containing index.html) served on every path not used by the API.
        StaticDir string
}

// NewHandler returns the API's root handler: all routes wrapped in CORS.
func NewHandler(cfg Config) http.Handler {
        mux := http.NewServeMux()

        mux.HandleFunc("GET /health", handleHealth)
        mux.HandleFunc("GET /api/v1/operations", handleOperations)
        mux.HandleFunc("POST /api/v1/calculate", handleCalculate)

        // Method-less patterns for the same paths. The mux prefers the
        // method-specific patterns above, so these only catch wrong methods and
        // let us return 405 in the shared JSON error shape (the mux's built-in
        // 405 is plain text).
        mux.Handle("/health", methodNotAllowed("GET, HEAD"))
        mux.Handle("/api/v1/operations", methodNotAllowed("GET, HEAD"))
        mux.Handle("/api/v1/calculate", methodNotAllowed("POST"))

        // Unknown API paths are always a JSON 404, even when serving the frontend.
        mux.HandleFunc("/api/", handleNotFound)
        mux.HandleFunc("/health/", handleNotFound)

        // Everything else is the frontend if configured, otherwise a JSON 404.
        if cfg.StaticDir != "" {
                mux.Handle("/", staticHandler(cfg.StaticDir))
        } else {
                mux.HandleFunc("/", handleNotFound)
        }

        return cors(cfg.AllowedOrigin, mux)
}

// handleNotFound answers unknown paths with a JSON 404.
func handleNotFound(w http.ResponseWriter, r *http.Request) {
        writeError(w, &apiError{http.StatusNotFound, CodeNotFound, "no route for " + r.URL.Path})
}

// methodNotAllowed answers a known path called with the wrong method.
func methodNotAllowed(allow string) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("Allow", allow)
                writeError(w, &apiError{http.StatusMethodNotAllowed, CodeMethodNotAllowed,
                        "method " + r.Method + " is not allowed; use " + allow})
        })
}

// cors adds CORS headers for allowedOrigin and answers preflight requests.
// Requests from other origins get no CORS headers, so browsers block them.
func cors(allowedOrigin string, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                h := w.Header()
                h.Add("Vary", "Origin")

                origin := r.Header.Get("Origin")
                allowed := origin != "" && (allowedOrigin == "*" || origin == allowedOrigin)
                if allowed {
                        if allowedOrigin == "*" {
                                h.Set("Access-Control-Allow-Origin", "*")
                        } else {
                                h.Set("Access-Control-Allow-Origin", origin)
                        }
                }

                isPreflight := r.Method == http.MethodOptions && r.Header.Get("Access-Control-Request-Method") != ""
                if isPreflight {
                        if allowed {
                                h.Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
                                h.Set("Access-Control-Allow-Headers", "Content-Type")
                                h.Set("Access-Control-Max-Age", "600")
                        }
                        w.WriteHeader(http.StatusNoContent)
                        return
                }
                next.ServeHTTP(w, r)
        })
}

import (
        "io"
        "io/fs"
        "log/slog"
        "net/http"
        "os"
        "path"
        "strings"
)

// staticHandler serves the built single-page app from dir:
//   - existing files are served as-is; Vite's hashed files under assets/
//     are cached for a year, since their names change when their content does;
//   - any other path (including directories) gets index.html, so client-side
//     routes work on reload and directory listings are never exposed;
//   - only GET and HEAD are allowed.
func staticHandler(dir string) http.Handler {
        fsys := os.DirFS(dir)
        wrongMethod := methodNotAllowed("GET, HEAD")

        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if r.Method != http.MethodGet && r.Method != http.MethodHead {
                        wrongMethod.ServeHTTP(w, r)
                        return
                }

                // path.Clean on a rooted path removes any "..", so name stays inside dir
                // (os.DirFS also rejects names that escape it).
                name := strings.TrimPrefix(path.Clean("/"+r.URL.Path), "/")
                if info, err := fs.Stat(fsys, name); err == nil && !info.IsDir() {
                        if strings.HasPrefix(name, "assets/") {
                                w.Header().Set("Cache-Control", "public, max-age=31536000, immutable")
                        } else {
                                w.Header().Set("Cache-Control", "no-cache")
                        }
                        serveFile(w, r, fsys, name)
                        return
                }

                // index.html must always be revalidated so new deploys are picked up.
                w.Header().Set("Cache-Control", "no-cache")
                serveFile(w, r, fsys, "index.html")
        })
}

// serveFile writes one file from fsys. http.ServeContent sets Content-Type
// from the extension and handles HEAD, Range and conditional requests.
func serveFile(w http.ResponseWriter, r *http.Request, fsys fs.FS, name string) {
        f, err := fsys.Open(name)
        if err != nil {
                handleNotFound(w, r)
                return
        }
        defer f.Close()

        info, err := f.Stat()
        content, seekable := f.(io.ReadSeeker)
        if err != nil || !seekable {
                slog.Error("cannot serve static file", "name", name, "err", err)
                writeError(w, &apiError{http.StatusInternalServerError, CodeInternal, "internal server error"})
                return
        }
        http.ServeContent(w, r, name, info.ModTime(), content)
}

// Package calculator provides pure arithmetic operations on float64 values.
//
// Every operation either returns a finite result and a nil error, or returns
// 0 and one of the sentinel errors declared below. Callers should compare
// errors with errors.Is.
package calculator

import (
        "errors"
        "math"
)

// Sentinel errors returned by the operations in this package.
var (
        // ErrDivisionByZero is returned when the divisor is zero.
        ErrDivisionByZero = errors.New("division by zero")

        // ErrNegativeSqrt is returned when taking the square root of a negative number.
        ErrNegativeSqrt = errors.New("square root of a negative number")

        // ErrNonFinite is returned when a result is NaN or ±Inf, for example on
        // overflow or when an input is already non-finite.
        ErrNonFinite = errors.New("result is not a finite number")
)

// Add returns a + b.
func Add(a, b float64) (float64, error) {
        return finite(a + b)
}

// Subtract returns a - b.
func Subtract(a, b float64) (float64, error) {
        return finite(a - b)
}

// Multiply returns a * b.
func Multiply(a, b float64) (float64, error) {
        return finite(a * b)
}

// Divide returns a / b. It returns ErrDivisionByZero when b is zero.
func Divide(a, b float64) (float64, error) {
        if b == 0 {
                return 0, ErrDivisionByZero
        }
        return finite(a / b)
}

// Power returns base raised to exp. Results that are not real and finite,
// such as a negative base with a fractional exponent or 0 raised to a
// negative power, return ErrNonFinite.
func Power(base, exp float64) (float64, error) {
        return finite(math.Pow(base, exp))
}

// Sqrt returns the square root of x. It returns ErrNegativeSqrt when x < 0.
func Sqrt(x float64) (float64, error) {
        if x < 0 {
                return 0, ErrNegativeSqrt
        }
        return finite(math.Sqrt(x))
}

// Percentage returns a percent of b, that is a*b/100.
// For example, Percentage(20, 50) returns 10.
func Percentage(a, b float64) (float64, error) {
        return finite(a * b / 100)
}

// finite returns v unchanged if it is a finite number, or ErrNonFinite otherwise.
func finite(v float64) (float64, error) {
        if math.IsNaN(v) || math.IsInf(v, 0) {
                return 0, ErrNonFinite
        }
        return v, nil
}
